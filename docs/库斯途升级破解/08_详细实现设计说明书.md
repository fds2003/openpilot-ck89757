## 二、高德车机广播解码模块实现设计

### 2.1 广播接收与解析逻辑 (`AmapBroadcastDecoder.kt`)

```kotlin
package tech.touka.amapnavi.data.amap

import android.content.BroadcastReceiver
import android.content.Context
import android.content.Intent
import tech.touka.amapnavi.core.model.NaviState
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow

class AmapBroadcastDecoder : BroadcastReceiver() {
    companion object {
        const val ACTION_AMAP_SEND = "AUTONAVI_STANDARD_BROADCAST_SEND"
        const val EXTRA_DRIVE_WAY = "EXTRA_DRIVE_WAY"
        const val EXTRA_ROAD_LIMIT_SPEED = "EXTRA_ROAD_LIMIT_SPEED"
        const val EXTRA_CAMERA_SPEED = "EXTRA_CAMERA_SPEED"
        const val EXTRA_CAMERA_DIST = "EXTRA_CAMERA_DIST"
        const val EXTRA_SEG_REMAIN_DIS = "EXTRA_SEG_REMAIN_DIS"
        const val EXTRA_TURN_TYPE = "EXTRA_TURN_TYPE"
        const val EXTRA_ROAD_CATE = "EXTRA_ROAD_CATE"
        const val EXTRA_TRAFFIC_LIGHT_SECONDS = "EXTRA_TRAFFIC_LIGHT_SECONDS"
        const val EXTRA_TRAFFIC_LIGHT_STATUS = "EXTRA_TRAFFIC_LIGHT_STATUS"
    }

    private val _naviStateFlow = MutableStateFlow(NaviState())
    val naviStateFlow: StateFlow<NaviState> = _naviStateFlow

    override fun onReceive(context: Context?, intent: Intent?) {
        if (intent == null || intent.action != ACTION_AMAP_SEND) return

        val driveWayStr = intent.getStringExtra(EXTRA_DRIVE_WAY) ?: ""
        val speedLimit = intent.getIntExtra(EXTRA_ROAD_LIMIT_SPEED, 0)
        val cameraSpeed = intent.getIntExtra(EXTRA_CAMERA_SPEED, 0)
        val cameraDist = intent.getIntExtra(EXTRA_CAMERA_DIST, 0)
        val tbtDist = intent.getIntExtra(EXTRA_SEG_REMAIN_DIS, 0)
        val turnType = intent.getIntExtra(EXTRA_TURN_TYPE, 0)
        val roadCate = intent.getIntExtra(EXTRA_ROAD_CATE, -1)
        val lightSec = intent.getIntExtra(EXTRA_TRAFFIC_LIGHT_SECONDS, -1)
        val lightStatus = intent.getIntExtra(EXTRA_TRAFFIC_LIGHT_STATUS, 0)

        // 解析推荐车道与禁行车道: 格式如 "s|s|S|X"
        val tokens = if (driveWayStr.isNotEmpty()) driveWayStr.split("|") else emptyList()
        val suggestList = mutableListOf<Int>()
        val disableList = mutableListOf<Int>()

        tokens.forEachIndexed { index, token ->
            if (token.isNotEmpty()) {
                val c = token[0]
                if (c == 'X') {
                    disableList.add(index)
                } else if (c.isUpperCase()) {
                    suggestList.add(index) // 大写字母表示推荐车道 (如 'S', 'L', 'R')
                }
            }
        }

        _naviStateFlow.value = _naviStateFlow.value.copy(
            navValid = true,
            roadSpeedLimit = speedLimit,
            cameraSpeedLimit = cameraSpeed,
            cameraDistance = cameraDist,
            tbtDistance = tbtDist,
            tbtTurnType = turnType,
            trafficLightSeconds = lightSec,
            trafficLightStatus = lightStatus,
            driveWayRaw = driveWayStr,
            suggestLanes = suggestList,
            disableLanes = disableList,
            isHighway = (roadCate == 0),
            timestamp = System.currentTimeMillis()
        )
    }
}
```

## 四、自主超车状态机详细设计与状态转移矩阵

### 4.1 状态转移矩阵 (State Transition Table)

| 当前状态 (`CurrentState`) | 触发条件 (Guard Condition) | 动作 (Action) | 下一状态 (`NextState`) |
| :--- | :--- | :--- | :--- |
| **IDLE (空闲)** | 车辆已接管 (`engaged=true`) && 车速 $\ge 60$ && 高速路段 | 启动前车监控 | **EVALUATING** |
| **COOLDOWN (冷却)** | 当前时间 $\ge$ 冷却到期时间戳 | 恢复超车准入 | **IDLE** |
| **EVALUATING (评估)** | 前车时距 $10\text{m} \le drel \le 60\text{m}$ 且 $(\Delta v \ge 10 \text{ 或 } \text{Ratio} \le 0.88)$ | 锁定前车压速 | **SIDE_CHECKING** |
| **SIDE_CHECKING (侧查)** | 目标侧标线为虚线 && BSD 无车 && 激光时距安全 | 播报预备音 `audio_pre_lane_left` | **PREPARE** |
| **SIDE_CHECKING (侧查)** | 标线为实线 | 播报阻挡音 `audio_solid_blocked` | **IDLE** |
| **SIDE_CHECKING (侧查)** | BSD 报警 或 激光时距不足 | 播报警告音 `audio_car_watchout` | **IDLE** |
| **PREPARE (准备)** | 发送前二次校验完全通过 | 下发 `LANECHANGE` 指令，播报变道音 | **EXECUTING** |
| **EXECUTING (执行)** | 驾驶员反向强打方向盘 ($|\Delta steer| > 25^{\circ}$) | 撤销指令，播报取消音，锁定 60 秒 | **COOLDOWN** |
| **EXECUTING (执行)** | 变道完成反馈 (`atc_state == 3`) 或 执行超时 ($>4.5\text{s}$) | 播报完成音 `audio_lane_change_ok`，锁定 20 秒 | **COOLDOWN** |

## 五、侧向防撞与激光雷达净空评估算法实现设计

### 5.1 动态制动安全净空算法实现 (`LateralSafetyEvaluator.kt`)

```kotlin
package tech.touka.amapnavi.domain.safety

import tech.touka.amapnavi.core.model.CarState

data class SafetyAuditResult(
    val isSafe: Boolean,
    val isSolidBlocked: Boolean = false,
    val isBsdBlocked: Boolean = false,
    val isLidarBlocked: Boolean = false,
    val reasonDesc: String = ""
)

class LateralSafetyEvaluator(
    private val minSafeTimeHeadway: Float = 2.0f, // 最小安全时距 (秒)
    private val vrelExtrapolationTime: Float = 4.0f // 相对速度外推时距 (秒)
) {
    fun evaluateSide(direction: String, car: CarState, isTargetSolid: Boolean): SafetyAuditResult {
        // 1. 标线准入法规检查
        if (isTargetSolid) {
            return SafetyAuditResult(isSafe = false, isSolidBlocked = true, reasonDesc = "侧向为实线禁止变道")
        }

        // 2. 原车毫米波盲区检查
        val bsdAlarm = if (direction == "LEFT") car.leftBlindspot else car.rightBlindspot
        if (bsdAlarm) {
            return SafetyAuditResult(isSafe = false, isBsdBlocked = true, reasonDesc = "毫米波盲区有车")
        }

        // 3. 激光雷达盲区标志检查
        val lidarBlind = if (direction == "LEFT") car.lidarLeftRearBlind else car.lidarRightRearBlind
        if (lidarBlind) {
            return SafetyAuditResult(isSafe = false, isLidarBlocked = true, reasonDesc = "激光雷达侧后盲区报警")
        }

        // 4. 激光雷达动态制动净空模型计算
        val actualRearDist = if (direction == "LEFT") car.leftRearDistance else car.rightRearDistance
        val vEgoMs = car.vEgo / 3.6f

        // 计算动态所需后方安全净空: D_safe = vEgo * T_headway + max(0, -vrel) * T_vrel
        val vrelRear = car.lidarBehindVrelTime // 逼近时距
        val requiredDistance = (vEgoMs * minSafeTimeHeadway).coerceAtLeast(10.0f)

        if (actualRearDist < requiredDistance) {
            return SafetyAuditResult(
                isSafe = false,
                isLidarBlocked = true,
                reasonDesc = "侧后方车距过近(实际:${actualRearDist}m < 门限:${requiredDistance}m)"
            )
        }

        return SafetyAuditResult(isSafe = true, reasonDesc = "侧向安全允许变道")
    }
}
```

## 六、匝道平滑减速曲线规划器实现设计

### 6.1 匀减速规划数学模型 (`RampSpeedPlanner.kt`)

当自车在高速公路上临近匝道分岔路口时，为避免临近匝道急刹，系统采用**平滑匀减速模型**规划瞬时目标速度：

$$v(s) = \sqrt{v_{target}^2 + 2 \cdot a_{decel} \cdot s}$$

其中：
- $s$：当前距分岔路口的剩余距离（$s = \text{tbtDistance}$）；
- $v_{target}$：匝道目标车速（例如 $60\text{km/h} = 16.67\text{m/s}$）；
- $a_{decel}$：舒适减速加速度（设定为 $0.8 \sim 1.2\text{m/s}^2$）。

```kotlin
package tech.touka.amapnavi.domain.decision

class RampSpeedPlanner(
    private val rampTargetSpeedKmh: Int = 60,
    private val decelAcceleration: Float = 1.0f // 舒适减速度 1.0 m/s^2
) {
    fun calculateCruiseSpeed(currentDistM: Int, currentCruiseKmh: Int): Int {
        if (currentDistM > 500) return currentCruiseKmh

        val targetSpeedMs = rampTargetSpeedKmh / 3.6f
        val vCurrentMs = currentCruiseKmh / 3.6f

        // 计算当前剩余距离下的物理允许车速
        val allowedSpeedSq = targetSpeedMs * targetSpeedMs + 2 * decelAcceleration * currentDistM
        val allowedSpeedMs = Math.sqrt(allowedSpeedSq.toDouble()).toFloat()
        val plannedKmh = (allowedSpeedMs * 3.6f).toInt()

        // 仅在规划值低于当前设定巡航值时实施降速
        return plannedKmh.coerceIn(rampTargetSpeedKmh, currentCruiseKmh)
    }
}
```

## 七、多通道低延迟音效反馈引擎实现设计

针对 Android 4.4 车机音频系统延迟大的问题，**必须废弃 `MediaPlayer`，全面采用 `SoundPool`** 进行内存预加载，实现 $<10\text{ms}$ 瞬时发声：

```kotlin
package tech.touka.amapnavi.ui.sound

import android.content.Context
import android.media.AudioManager
import android.media.SoundPool

class SoundFeedbackEngine(private val context: Context) {
    private val soundPool: SoundPool = SoundPool(4, AudioManager.STREAM_MUSIC, 0)
    private val soundMap = mutableMapOf<String, Int>()

    init {
        preload("audio_pre_lane_left.wav")
        preload("audio_pre_lane_right.wav")
        preload("audio_lane_change.wav")
        preload("audio_lane_change_ok.wav")
        preload("audio_lane_change_cancel.wav")
        preload("audio_solid_blocked.wav")
        preload("audio_car_watchout.wav")
        preload("traffic_sign_green.wav")
    }

    private fun preload(fileName: String) {
        val resId = context.resources.getIdentifier(
            fileName.removeSuffix(".wav"), "raw", context.packageName
        )
        if (resId != 0) {
            val soundId = soundPool.load(context, resId, 1)
            soundMap[fileName] = soundId
        }
    }

    fun play(fileName: String) {
        val soundId = soundMap[fileName] ?: return
        soundPool.play(soundId, 1.0f, 1.0f, 1, 0, 1.0f)
    }

    fun release() {
        soundPool.release()
    }
}
```
