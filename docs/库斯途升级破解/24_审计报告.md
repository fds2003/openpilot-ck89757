B-1（R-05）CarrotTelemetryParser.kt — 用预常量 key + 手写数值解析消除每字段的 String 分配：

    object CarrotTelemetryParser {
        private const val K_V_EGO = "\"v_ego_kph\":"
        private const val K_CRUISE = "\"v_cruise_kph\":"
        private const val K_STEER = "\"steer_angle\":"
        private const val K_DREL = "\"drel\":"
        private const val K_VREL = "\"vrel\":"
        private const val K_TTC = "\"lidar_behind_vdist_time\":"

        fun parse(payload: String): CarState = CarState(
            vEgo = scanInt(payload, K_V_EGO, scanInt(payload, "\"v_ego\":", 0)),
            vCruise = scanInt(payload, K_CRUISE, 0),
            steerAngle = scanFloat(payload, K_STEER, 0.0f),
            drel = scanFloat(payload, K_DREL, 0.0f),
            vrel = scanFloat(payload, K_VREL, 0.0f),
            lidarBehindVdistTime = scanFloat(payload, K_TTC, 0.0f),
            // ...其余字段同法替换 fastGetBool/fastGetInt/fastGetFloat
        )

        /** 不使用 substring / toXxx，逐字符累加，零中间对象分配 */
        private fun scanFloat(s: String, key: String, def: Float): Float {
            val i = s.indexOf(key)
            if (i < 0) return def
            var p = i + key.length
            while (p < s.length && (s[p] == ' ' || s[p] == '\t')) p++
            var neg = false
            if (p < s.length && s[p] == '-') { neg = true; p++ }
            var v = 0f; var div = 1f; var dot = false; var any = false
            while (p < s.length) {
                val c = s[p]
                if (c in '0'..'9') {
                    any = true
                    if (dot) { div *= 10f; v += (c - '0') / div }
                    else v = v * 10f + (c - '0')
                } else if (c == '.' && !dot) { dot = true }
                else break
                p++
            }
            return if (any) (if (neg) -v else v) else def
        }

        private fun scanInt(s: String, key: String, def: Int): Int =

## 四、极端边缘工况推演与反例压力测试

场景 A（高负荷内存极限 + R 挡倒车）：高德三维立交渲染占 209MB，同时 20Hz 遥测风暴（补丁前每包 600+ 次/秒 String 分配）触发 Dalvik GC_FOR_ALLOC STW 30~80ms；叠加 22 §3.4 自述的峰值 35~55MB（已越过 §1.2 的 50MB LMK 红线），原厂 reverse_service 需在 0.5s 内抢屏，GC 停顿与 LMK 回收恰好破坏该时序。当前 R-05/R-10 使"140MB 安全缓冲带"不可证；补齐 B-1 与完整 dumpsys 原始输出前，该场景不能判定为安全。

场景 B（110km/h 变道中热点断电 + 后车快车）：端口契约反转使车机指令根本到不了 C3（R-01），本次变道更可能由 C3 自主逻辑触发、完全不受车机监控。变道一旦进入 laneChangeStarting，TTC 不再复检（R-16）；无心跳超时，carrot_lane_change_count 从 400 帧自然衰减，期间沿用上次的 lane_change_available 与 overtake_mode；归零后进入 else 分支的常规自主变道逻辑。即断网不仅不会"1 秒降级"，反而可能在后车逼近时继续切线 —— 与 22 §5.2 结论完全相反。

场景 C（雨雪点云跳变，TTC 99→0.1→99）：当前 rear_conflict = (0.0 < ttc < thr)，0.1s 瞬时阻断、恢复 99 立即放行，无连续性/滤波/去抖，单点跳变即可在两帧间翻转决策。更危险的是 main_lb_drel 负值→ttc 负→放行、vrel≈0 并排→ttc=99→放行、capnp 字段默认 0.0→消息缺失→放行（R-15）。补丁 C-1 把 ttc<=0 统一判为阻断后才具备 fail-safe 语义，但仍需加 N 帧连续确认以抑制抖动。
